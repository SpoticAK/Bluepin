   798	function BiomarkerRow({ biomarker, history, onSelectBiomarker }: { key?: React.Key, biomarker: Biomarker, history: any[], onSelectBiomarker: (b: Biomarker, h: any[]) => void }) {
   799	 const isTier1 = TIER_1.includes(biomarker.name.toLowerCase().trim());
   800	 const canShowGraph = isTier1 || history.length > 1;
   801	 const canOpenModal = canShowGraph || !!biomarker.info;
   802	
   803	 let finalStatus = biomarker.status; 
   804	 if (!['Healthy', 'Needs Attention', 'Borderline'].includes(finalStatus as string)) {
   805	 finalStatus = 'Healthy'; 
   806	 }
   807	 
   808	 let diffString = null;
   809	 let diffPrefix = '';
   810	 if (history.length > 1) {
   811	 const prevVal = Number(history[history.length - 2].value);
   812	 const curVal = Number(biomarker.value);
   813	 if (!isNaN(prevVal) && !isNaN(curVal)) {
   814	 const diff = curVal - prevVal;
   815	 if (diff !== 0) {
   816	 diffPrefix = diff > 0 ? '↑' : '↓';
   817	 diffString = `${diffPrefix} ${Math.abs(diff).toFixed(diff % 1 !== 0 ? 1 : 0)}`;
   818	 }
   819	 }
   820	 }
   821	
   822	 return (
   823	 <div 
   824	 onClick={() => { if (canOpenModal) onSelectBiomarker(biomarker, history) }}
   825	 className={cn(
   826	 "p-3 sm:p-4 mx-2 rounded-2xl flex flex-row items-center justify-between group transition-colors mb-1", 
   827	 finalStatus === 'Healthy' ? "bg-transparent hover:bg-theme-card-sec" : 
   828	 finalStatus === 'Needs Attention' ? "bg-theme-critical/5 hover:bg-theme-critical/10" : 
   829	 "bg-theme-warning/5 hover:bg-theme-warning/10",
   830	 canOpenModal ? "cursor-pointer" : "cursor-default"
   831	 )}
   832	 >
   833	 <div className="flex-1 flex flex-col justify-center">
   834	 <h4 className="font-bold text-theme-text text-sm sm:text-base leading-tight flex items-center gap-2">
   835	 {biomarker.name}
   836	 {biomarker.info && (
   837	 <Info size={14} className="text-theme-text-sec shrink-0" />
   838	 )}
   839	 </h4>
   840	 {(biomarker.refRangeText || (biomarker.refMin != null || biomarker.refMax != null)) && (
   841	 <span className="text-xs font-medium text-theme-text-sec mt-0.5">
   842	 Ref: {biomarker.refRangeText || (biomarker.refMin === 0 && biomarker.refMax === 0 ? 'Not specified' : `${biomarker.refMin ?? '?'} - ${biomarker.refMax ?? '?'}`)}
   843	 </span>
   844	 )}
   845	 </div>
   846
